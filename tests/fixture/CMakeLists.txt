cmake_minimum_required(VERSION 3.22)
project(BoostPackFixture LANGUAGES CXX)

include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/BoostPack.cmake")
# Real CMake targets are configured before their Boost:: dependencies exist.
# Verify that discovery follows the configured graph, including nested files,
# variable expansion, platform conditions and optional target expressions.
set(fake_modules config bar headers foo pfr other includes condition dynamic nested optional support nested_support genex genex_inactive_unknown genex_bad genex_bad_condition genex_config_union genex_config_exclusive genex_target_exists missing_target)
foreach(module IN LISTS fake_modules)
  set(dir "${CMAKE_CURRENT_BINARY_DIR}/source/${module}")
  file(MAKE_DIRECTORY "${dir}")
  set(definition "add_library(boost_${module} INTERFACE)\n")
  if(NOT module STREQUAL "missing_target")
    string(APPEND definition "add_library(Boost::${module} ALIAS boost_${module})\n")
  endif()
  if(module STREQUAL "foo")
    string(APPEND definition
      "# The commented Boost::ghost must not be fetched\n"
      "target_link_libraries(boost_foo INTERFACE Boost::bar Boost::headers)\n"
      "add_library(boost_foo_detail INTERFACE)\n"
      "add_library(Boost::foo_detail ALIAS boost_foo_detail)\n")
  elseif(module STREQUAL "bar")
    if(TEST_CASE STREQUAL "cycle")
      string(APPEND definition "target_link_libraries(boost_bar INTERFACE Boost::foo)\n")
    else()
      string(APPEND definition "target_link_libraries(boost_bar INTERFACE Boost::config)\n")
    endif()
  elseif(module STREQUAL "includes")
    file(WRITE "${dir}/deps.cmake" "target_link_libraries(boost_includes INTERFACE Boost::bar)\n")
    string(APPEND definition "include(\"\${CMAKE_CURRENT_LIST_DIR}/deps.cmake\")\n")
  elseif(module STREQUAL "condition")
    string(APPEND definition
      "if(TEST_CASE STREQUAL conditional)\n"
      "  target_link_libraries(boost_condition INTERFACE Boost::bar)\n"
      "else()\n"
      "  target_link_libraries(boost_condition INTERFACE Boost::ghost)\n"
      "endif()\n")
  elseif(module STREQUAL "dynamic")
    string(APPEND definition
      "set(dependency bar)\n"
      "target_link_libraries(boost_dynamic INTERFACE Boost::\${dependency})\n")
  elseif(module STREQUAL "nested")
    file(MAKE_DIRECTORY "${dir}/impl")
    file(WRITE "${dir}/impl/CMakeLists.txt" "add_library(boost_nested_impl INTERFACE)\ntarget_link_libraries(boost_nested_impl INTERFACE Boost::bar)\n")
    string(APPEND definition "add_subdirectory(impl)\ntarget_link_libraries(boost_nested INTERFACE boost_nested_impl)\n")
  elseif(module STREQUAL "optional")
    string(APPEND definition "target_link_libraries(boost_optional INTERFACE \"$<TARGET_NAME_IF_EXISTS:Boost::ghost>\" Boost::bar)\n")
  elseif(module STREQUAL "genex")
    string(APPEND definition
      "target_link_libraries(boost_genex INTERFACE\n"
      "  \"$<IF:$<BOOL:1>,Boost::bar,Boost::ghost_true>\"\n"
      "  \"$<IF:$<BOOL:0>,Boost::ghost_false,Boost::headers>\"\n"
      "  \"$<$<NOT:$<BOOL:0>>:Boost::pfr>\"\n"
      "  \"$<IF:$<AND:$<BOOL:1>,$<NOT:$<BOOL:0>>>,Boost::config,Boost::ghost_and>\"\n"
      "  \"$<BUILD_INTERFACE:$<LINK_ONLY:Boost::pfr>>\"\n"
      "  \"$<INSTALL_INTERFACE:Boost::ghost_install>\"\n"
      "  \"$<TARGET_NAME_IF_EXISTS:Boost::ghost_optional>\"\n"
      "  \"$<IF:$<CONFIG:Debug>,Boost::other,Boost::headers>\"\n"
      "  \"$<IF:$<PLATFORM_ID:${CMAKE_SYSTEM_NAME}>,Boost::config,Boost::ghost_platform>\"\n"
      ")\n")
  elseif(module STREQUAL "genex_inactive_unknown")
    string(APPEND definition
      "target_link_libraries(boost_genex_inactive_unknown INTERFACE\n"
      "  \"$<IF:$<BOOL:1>,Boost::bar,$<JOIN:Boost::ghost_else,_>>\"\n"
      "  \"$<IF:$<AND:0,$<BOOL:$<JOIN:Boost::ghost_cond,_>>>,Boost::ghost_true,Boost::headers>\"\n"
      "  \"$<INSTALL_INTERFACE:$<JOIN:Boost::ghost_install,_>>\"\n"
      "  \"$<0:$<JOIN:Boost::ghost_zero,_>>\"\n"
      ")\n")
  elseif(module STREQUAL "genex_config_union")
    string(APPEND definition
      "target_link_libraries(boost_genex_config_union INTERFACE \"$<IF:$<CONFIG:Debug>,Boost::bar,Boost::headers>\")\n")
  elseif(module STREQUAL "genex_config_exclusive")
    string(APPEND definition
      "target_link_libraries(boost_genex_config_exclusive INTERFACE \"$<IF:$<AND:$<CONFIG:Debug>,$<CONFIG:Release>>,Boost::ghost,Boost::bar>\")\n")
  elseif(module STREQUAL "genex_target_exists")
    string(APPEND definition
      "target_link_libraries(boost_genex_target_exists INTERFACE \"$<IF:$<TARGET_EXISTS:Boost::bar>,Boost::bar,Boost::headers>\")\n")
  elseif(module STREQUAL "genex_bad")
    string(APPEND definition
      "target_link_libraries(boost_genex_bad INTERFACE \"$<JOIN:Boost::bar,->\")\n")
  elseif(module STREQUAL "genex_bad_condition")
    string(APPEND definition
      "target_link_libraries(boost_genex_bad_condition INTERFACE \"$<IF:$<BOOL:$<JOIN:1,_>>,Boost::bar,Boost::headers>\")\n")
  elseif(module STREQUAL "support")
    string(APPEND definition
      "target_link_libraries(boost_support INTERFACE Boost::config)\n"
      "boost_install(TARGETS boost_support VERSION \${BOOST_SUPERPROJECT_VERSION})\n")
  elseif(module STREQUAL "nested_support")
    file(MAKE_DIRECTORY "${dir}/cmake" "${dir}/impl")
    file(WRITE "${dir}/cmake/helpers.cmake"
      "boost_install(TARGETS boost_nested_support VERSION \${BOOST_SUPERPROJECT_VERSION})\n")
    file(WRITE "${dir}/impl/CMakeLists.txt"
      "include(BoostInstall)\n")
    string(APPEND definition
      "include(${dir}/cmake/helpers.cmake)\n"
      "add_subdirectory(impl)\n")
  endif()
  file(WRITE "${dir}/CMakeLists.txt" "${definition}")
endforeach()

# Pretend the version-matched boostorg/cmake archive is available locally.
set(support_dir "${CMAKE_CURRENT_BINARY_DIR}/source/cmake")
file(MAKE_DIRECTORY "${support_dir}/include")
file(WRITE "${support_dir}/include/BoostInstall.cmake" [=[
function(boost_install)
  if(NOT BOOST_SKIP_INSTALL_RULES)
    message(FATAL_ERROR "BoostPack must disable upstream install rules")
  endif()
  set_property(GLOBAL APPEND PROPERTY BOOSTPACK_TEST_SUPPORT_CALLED yes)
endfunction()
]=])
# Build local Boost-shaped archives so every configuration test is self-contained.
set(fixture_mirror "${CMAKE_CURRENT_BINARY_DIR}/mirror")
set(archive_modules ${fake_modules} cmake)
foreach(module IN LISTS archive_modules)
  file(MAKE_DIRECTORY "${fixture_mirror}/${module}")
  execute_process(COMMAND "${CMAKE_COMMAND}" -E tar czf
    "${fixture_mirror}/${module}/source.tar.gz" "${module}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/source"
    RESULT_VARIABLE archive_result)
  if(NOT archive_result EQUAL 0)
    message(FATAL_ERROR "Failed creating fixture archive ${module}")
  endif()
  foreach(version IN ITEMS 1.76.0 1.80.0 1.85.0 1.90.0 1.91.0 1.92.0)
    file(COPY_FILE "${fixture_mirror}/${module}/source.tar.gz"
      "${fixture_mirror}/${module}/boost-${version}.tar.gz")
    file(COPY_FILE "${fixture_mirror}/${module}/source.tar.gz"
      "${fixture_mirror}/${module}/source-boost-${version}.tar.gz")
  endforeach()
endforeach()
if(WIN32)
  set(fixture_repository "file:///${fixture_mirror}")
else()
  set(fixture_repository "file://${fixture_mirror}")
endif()
function(_fixture_boostpack_add)
  set(args ${ARGN})
  list(FIND args REPOSITORY repo_index)
  if(repo_index EQUAL -1)
    list(APPEND args REPOSITORY "${fixture_repository}")
  endif()
  list(FIND args URL_TEMPLATE template_index)
  if(template_index EQUAL -1)
    list(APPEND args URL_TEMPLATE "@REPOSITORY@/@MODULE@/@TAG@.tar.gz")
  endif()
  boostpack_add(${args})
endfunction()

if(TEST_CASE STREQUAL "minimal")
  _fixture_boostpack_add(VERSION 1.85.0 COMPONENTS pfr)
  get_property(modules GLOBAL PROPERTY BOOSTPACK_MODULES)
  if(NOT modules STREQUAL "pfr")
    message(FATAL_ERROR "Expected only pfr, got '${modules}'")
  endif()
elseif(TEST_CASE STREQUAL "closure")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS foo)
  get_property(modules GLOBAL PROPERTY BOOSTPACK_MODULES)
  list(SORT modules)
  if(NOT modules STREQUAL "bar;config;foo;headers")
    message(FATAL_ERROR "Incorrect closure: ${modules}")
  endif()
elseif(TEST_CASE STREQUAL "incremental")
  _fixture_boostpack_add(VERSION 1.76.0 COMPONENTS pfr)
  _fixture_boostpack_add(VERSION 1.76.0 COMPONENTS foo pfr)
  get_property(modules GLOBAL PROPERTY BOOSTPACK_MODULES)
  list(LENGTH modules length)
  if(NOT length EQUAL 5)
    message(FATAL_ERROR "Expected five unique targets, got ${modules}")
  endif()
elseif(TEST_CASE STREQUAL "reinclude")
  include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/BoostPack.cmake")
  _fixture_boostpack_add(VERSION 1.80.0 COMPONENTS pfr)
  _fixture_boostpack_add(VERSION 1.80.0 COMPONENTS pfr)
elseif(TEST_CASE STREQUAL "repository")
  _fixture_boostpack_add(VERSION 1.90.0 COMPONENTS foo
    REPOSITORY "${fixture_repository}"
    URL_TEMPLATE "@REPOSITORY@/@MODULE@/@TAG@.tar.gz")
  get_property(repo GLOBAL PROPERTY BOOSTPACK_REPOSITORY)
  if(NOT repo STREQUAL fixture_repository)
    message(FATAL_ERROR "Repository was not honored: ${repo}")
  endif()
elseif(TEST_CASE STREQUAL "template")
  _fixture_boostpack_add(VERSION 1.92.0
    REPOSITORY "${fixture_repository}"
    URL_TEMPLATE "@REPOSITORY@/@MODULE@/source-@TAG@.tar.gz"
    COMPONENTS foo)
elseif(TEST_CASE STREQUAL "conditional")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS condition)
  if(TARGET Boost::ghost OR NOT TARGET Boost::bar)
    message(FATAL_ERROR "Inactive branch must not contribute a dependency")
  endif()
elseif(TEST_CASE STREQUAL "included")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS includes)
  if(NOT TARGET Boost::bar)
    message(FATAL_ERROR "Include-based dependency was not discovered")
  endif()
elseif(TEST_CASE STREQUAL "dynamic")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS dynamic)
  if(NOT TARGET Boost::bar)
    message(FATAL_ERROR "Variable-based dependency was not discovered")
  endif()
elseif(TEST_CASE STREQUAL "nested")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS nested)
  if(NOT TARGET Boost::bar)
    message(FATAL_ERROR "Nested target dependency was not discovered")
  endif()
elseif(TEST_CASE STREQUAL "optional")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS optional)
  if(TARGET Boost::ghost OR NOT TARGET Boost::bar)
    message(FATAL_ERROR "Optional dependency incorrectly fetched")
  endif()
elseif(TEST_CASE STREQUAL "support")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS support)
  get_property(calls GLOBAL PROPERTY BOOSTPACK_TEST_SUPPORT_CALLED)
  if(NOT calls STREQUAL yes OR NOT TARGET Boost::config)
    message(FATAL_ERROR "Upstream support was not initialized")
  endif()
elseif(TEST_CASE STREQUAL "nested_support")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS nested_support)
  get_property(calls GLOBAL PROPERTY BOOSTPACK_TEST_SUPPORT_CALLED)
  if(NOT calls STREQUAL yes OR NOT TARGET Boost::nested_support)
    message(FATAL_ERROR "Nested upstream CMake helpers were not initialized: ${calls}")
  endif()
elseif(TEST_CASE STREQUAL "genex")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS genex)
  foreach(expected IN ITEMS genex bar config headers pfr)
    if(NOT TARGET Boost::${expected})
      message(FATAL_ERROR "Missing generator-expression dependency ${expected}")
    endif()
  endforeach()
  foreach(unwanted IN ITEMS ghost_true ghost_false ghost_and ghost_install
      ghost_optional ghost_platform)
    if(TARGET Boost::${unwanted})
      message(FATAL_ERROR "Inactive generator-expression branch was downloaded: ${unwanted}")
    endif()
  endforeach()
  set(expect_debug FALSE)
  if(CMAKE_CONFIGURATION_TYPES)
    list(FIND CMAKE_CONFIGURATION_TYPES Debug debug_index)
    if(NOT debug_index EQUAL -1)
      set(expect_debug TRUE)
    endif()
  elseif(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(expect_debug TRUE)
  endif()
  if(expect_debug AND NOT TARGET Boost::other)
    message(FATAL_ERROR "A Debug configuration requires Boost::other")
  elseif(NOT expect_debug AND TARGET Boost::other)
    message(FATAL_ERROR "Inactive Debug-only dependency was downloaded")
  endif()
elseif(TEST_CASE STREQUAL "genex_inactive_unknown")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS genex_inactive_unknown)
  foreach(expected IN ITEMS genex_inactive_unknown bar config headers)
    if(NOT TARGET Boost::${expected})
      message(FATAL_ERROR "Missing dependency after inactive genex pruning: ${expected}")
    endif()
  endforeach()
  foreach(unwanted IN ITEMS ghost_else ghost_cond ghost_true ghost_install ghost_zero)
    if(TARGET Boost::${unwanted})
      message(FATAL_ERROR "Inactive genex over-fetched Boost::${unwanted}")
    endif()
  endforeach()
elseif(TEST_CASE STREQUAL "genex_config_union")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS genex_config_union)
  if(CMAKE_CONFIGURATION_TYPES)
    if(NOT TARGET Boost::bar OR NOT TARGET Boost::headers)
      message(FATAL_ERROR "Multi-config dependency union was not resolved")
    endif()
  endif()
elseif(TEST_CASE STREQUAL "genex_config_exclusive")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS genex_config_exclusive)
  if(TARGET Boost::ghost OR NOT TARGET Boost::bar)
    message(FATAL_ERROR "Mutually exclusive configurations over-fetched modules")
  endif()
elseif(TEST_CASE STREQUAL "genex_target_exists")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS genex_target_exists)
elseif(TEST_CASE STREQUAL "genex_bad")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS genex_bad)
elseif(TEST_CASE STREQUAL "genex_bad_condition")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS genex_bad_condition)
elseif(TEST_CASE STREQUAL "collision")
  add_library(foreign_bar INTERFACE)
  add_library(Boost::bar ALIAS foreign_bar)
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS foo)
elseif(TEST_CASE STREQUAL "version_conflict")
  _fixture_boostpack_add(VERSION 1.90.0 COMPONENTS pfr)
  _fixture_boostpack_add(VERSION 1.91.0 COMPONENTS foo)
elseif(TEST_CASE STREQUAL "repository_conflict")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS pfr)
  _fixture_boostpack_add(VERSION 1.92.0 REPOSITORY https://mirror.invalid/boost COMPONENTS foo)
elseif(TEST_CASE STREQUAL "template_conflict")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS pfr)
  _fixture_boostpack_add(VERSION 1.92.0
    URL_TEMPLATE "https://mirror.invalid/@MODULE@/@TAG@.tar.gz"
    COMPONENTS foo)
elseif(TEST_CASE STREQUAL "cycle")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS foo)
  if(NOT TARGET Boost::bar OR NOT TARGET Boost::foo)
    message(FATAL_ERROR "Interface dependency cycle lost a target")
  endif()
elseif(TEST_CASE STREQUAL "missing_offline")
  set(BOOSTPACK_OFFLINE ON)
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS foo)
elseif(TEST_CASE STREQUAL "missing_target")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS missing_target)
elseif(TEST_CASE STREQUAL "bad_module")
  _fixture_boostpack_add(VERSION 1.92.0 COMPONENTS ../evil)
elseif(TEST_CASE STREQUAL "bad_version")
  _fixture_boostpack_add(VERSION 1.92 COMPONENTS foo)
elseif(TEST_CASE STREQUAL "bad_template")
  _fixture_boostpack_add(VERSION 1.92.0
    URL_TEMPLATE "https://mirror.invalid/foo.tar.gz" COMPONENTS foo)
elseif(TEST_CASE STREQUAL "missing_arguments")
  _fixture_boostpack_add(VERSION 1.92.0)
else()
  message(FATAL_ERROR "Unknown fixture test case: ${TEST_CASE}")
endif()
